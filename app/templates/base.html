Collect customer leads directly from any website, landing page, backend service or terminal

Dual Authentication Required for All API Calls:

  • SECRET_KEY: uFNj1xDgFM9ONAtp (X-Secret-Key).
  • User API Key: Generated under Users & API Keys (X-API-Key). Default Platform Key: key_live_4cd32dfa2caa35276ca9f53446267f9c (Platform: Customer / portal@customer.com).

When leads are submitted with a User's API key, their platform name is automatically recorded in the From column.

API Endpoint (CORS Enabled)

POST https://your-domain.onrender.com/api/customers

Only phone is required (Indian 10-digit or +91 format, stored as +91XXXXXXXXXX, unique).

Payload Parameters Reference Content-Type: application/json
phone (required, string) 10 digits or +91XXXXXXXXXX (normalized & unique)
profile (optional, string/null) Image URL (http...) or null if not available
name, email, city, state (optional, string) Personal & geographical identity
address, landmark, pincode (optional, string) Address & postal pincode details

PHP (cURL Request) 
<?php
// PHP Customer Lead Submission to Customer Data API
$apiUrl = 'https://your-domain.onrender.com/api/customers';

// Dual Authentication Keys
$secretKey = 'uFNj1xDgFM9ONAtp';                          // Master SECRET_KEY (REQUIRED)
$apiKey    = 'key_live_4cd32dfa2caa35276ca9f53446267f9c'; // User API Key: Customer (portal@customer.com)

// Customer Data Payload
$payload = [
    'phone'    => '9876543210',        // REQUIRED: 10-digit Indian phone or +919876543210
    'name'     => 'Aarav Sharma',      // optional
    'email'    => 'aarav@example.com', // optional
    'city'     => 'Jaipur',            // optional
    'state'    => 'Rajasthan',         // optional
    'pincode'  => '302001',            // optional
    'profile'  => null                 // optional (Avatar image URL or null)
];

$ch = curl_init($apiUrl);
curl_setopt($ch, CURLOPT_POST, true);
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($payload));
curl_setopt($ch, CURLOPT_HTTPHEADER, [
    'Content-Type: application/json',
    'X-Secret-Key: ' . $secretKey,
    'X-API-Key: ' . $apiKey
]);

$response = curl_exec($ch);
$httpStatus = curl_getinfo($ch, CURLINFO_HTTP_CODE);

if (curl_errno($ch)) {
    echo 'cURL Error: ' . curl_error($ch);
} else {
    $result = json_decode($response, true);
    echo "HTTP Status: $httpStatus\n";
    print_r($result);
}
curl_close($ch);
?>

// Frontend Browser Website Lead Submission
async function submitCustomerLead(event) {
  if (event) event.preventDefault();

  const response = await fetch('/api/customers', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'X-Secret-Key': 'uFNj1xDgFM9ONAtp',                         // Master SECRET_KEY (REQUIRED)
      'X-API-Key': 'key_live_4cd32dfa2caa35276ca9f53446267f9c'   // User API Key: Customer (portal@customer.com)
    },
    body: JSON.stringify({
      phone: document.getElementById('phone')?.value || '9876543210', // REQUIRED
      name: document.getElementById('name')?.value || 'Aarav Sharma',
      email: document.getElementById('email')?.value || 'aarav@example.com',
      city: document.getElementById('city')?.value || 'Jaipur',
      state: document.getElementById('state')?.value || 'Rajasthan',
      pincode: document.getElementById('pincode')?.value || '302001',
      profile: null // Send null when image is not available
    })
  });

  const data = await response.json();
  if (response.ok) {
    alert('Thank you! Your information has been submitted.');
  } else {
    alert('Submission failed: ' + (data.detail || data.message || 'Error'));
  }
}
